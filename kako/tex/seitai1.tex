    \def\h{1.2} % 各段の高さ

    \node at (0, 0.5*\h) {\footnotesize 植物};
    
    % 中段：草食動物（一次消費者）
    \draw[fill=gray!25, thick] (-#2/2, \h) rectangle (#2/2, 2*\h);
    \node at (0, 1.5*\h) {\footnotesize 草食動物};

    \draw[fill=gray!40, thick] (-#1/2, 2*\h) rectangle (#1/2, 3*\h);
    \node at (0, 2.5*\h) {\footnotesize 肉食動物};
  \end{tikzpicture}%
}
